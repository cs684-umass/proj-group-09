def solve_question():
    # Step 1: Find the ICD9_CODE for the diagnosis "comp-oth vasc dev/graft" in the d_icd_diagnoses database
    icd9_code = SQLInterpreter("SELECT ICD9_CODE FROM d_icd_diagnoses WHERE SHORT_TITLE = 'comp-oth vasc dev/graft'")

    # Step 2: Find the HADM_ID associated with the ICD9_CODE in the diagnoses_icd database
    hadm_id = SQLInterpreter("SELECT HADM_ID FROM diagnoses_icd WHERE ICD9_CODE = '" + icd9_code + "'")

    # Step 3: Find the maximum total hospital cost associated with the HADM_ID in the cost database
